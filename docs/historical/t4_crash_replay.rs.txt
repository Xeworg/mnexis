use std::{
    env,
    io::{BufRead, BufReader, Write},
    process::{Command, Stdio},
    sync::{Arc, mpsc},
    thread,
    time::{Duration, Instant},
};

use arrow_array::{ArrayRef, FixedSizeListArray, RecordBatch, StringArray, types::Float32Type};
use arrow_schema::{DataType, Field, Schema};
use mnexis_core::open_sqlite;
use tempfile::tempdir_in;

const CHILD_MODE: &str = "MNEXIS_CRASH_REPLAY_CHILD";
const DATA_DIR: &str = "MNEXIS_CRASH_REPLAY_DATA_DIR";
const TABLE: &str = "t4_crash_replay";
const SQLITE_MARKER: &str = "sqlite-committed";
const LANCE_MARKER: &str = "lance-committed";
const READY_MARKER: &str = "T4_CRASH_REPLAY_READY";

/// Child fixture: complete both public store operations, announce readiness, then wait to be killed.
#[tokio::test]
#[ignore = "spawned by the parent crash-replay test"]
async fn crash_child_writes_then_waits_for_kill() {
    if env::var_os(CHILD_MODE).is_none() {
        return;
    }
    let data_dir = env::var_os(DATA_DIR).expect("parent must provide isolated data directory");
    let data_dir = std::path::PathBuf::from(data_dir);

    let sqlite_path = data_dir.join("mnexis.db");
    let mut connection = open_sqlite(&sqlite_path).expect("open spike SQLite database");
    connection
        .execute_batch("CREATE TABLE crash_markers (id TEXT PRIMARY KEY, value TEXT NOT NULL);")
        .expect("create SQLite marker table");
    let transaction = connection.transaction().expect("start SQLite transaction");
    transaction
        .execute(
            "INSERT INTO crash_markers (id, value) VALUES (?1, ?2)",
            [SQLITE_MARKER, "committed-before-kill"],
        )
        .expect("insert SQLite marker");
    transaction.commit().expect("commit SQLite marker");

    let vector_uri = data_dir.join("lancedb").to_string_lossy().into_owned();
    let db = lancedb::connect(vector_uri.as_str())
        .execute()
        .await
        .expect("connect to LanceDB");
    let _table = db
        .create_table(TABLE, vec![marker_batch(LANCE_MARKER)])
        .execute()
        .await
        .expect("create LanceDB table and await its write");

    println!("{READY_MARKER}");
    std::io::stdout()
        .flush()
        .expect("flush child readiness marker");
    loop {
        thread::park();
    }
}

/// Verifies only that writes whose APIs returned before the crash are visible after reopening.
/// It does not test in-flight writes, cross-store atomicity, or a general durability guarantee.
#[tokio::test]
async fn acknowledged_sqlite_and_lancedb_writes_survive_process_kill() {
    let directory = tempdir_in("target").expect("create isolated directory under target");
    let executable = env::current_exe().expect("locate this integration-test executable");
    let mut child = Command::new(executable)
        .arg("--exact")
        .arg("crash_child_writes_then_waits_for_kill")
        .arg("--ignored")
        .arg("--nocapture")
        .env(CHILD_MODE, "1")
        .env(DATA_DIR, directory.path())
        .stdin(Stdio::null())
        .stdout(Stdio::piped())
        .stderr(Stdio::null())
        .spawn()
        .expect("spawn isolated child process");

    let child_stdout = child.stdout.take().expect("capture child stdout");
    let (line_sender, line_receiver) = mpsc::channel();
    let reader = thread::spawn(move || {
        for line in BufReader::new(child_stdout).lines() {
            let Ok(line) = line else {
                break;
            };
            if line_sender.send(line).is_err() {
                break;
            }
        }
    });

    let deadline = Instant::now() + Duration::from_secs(60);
    let mut ready = false;
    while Instant::now() < deadline {
        match line_receiver.recv_timeout(Duration::from_millis(100)) {
            Ok(line) if line.contains(READY_MARKER) => {
                ready = true;
                break;
            }
            Ok(_) | Err(mpsc::RecvTimeoutError::Timeout) => {
                if child.try_wait().expect("check child status").is_some() {
                    break;
                }
            }
            Err(mpsc::RecvTimeoutError::Disconnected) => break,
        }
    }

    if !ready {
        let _ = child.kill();
        let status = child.wait().expect("reap child after failed readiness");
        reader.join().expect("join child output reader");
        panic!("child did not complete both writes before timeout; status: {status}");
    }

    child
        .kill()
        .expect("force-kill child after acknowledged writes");
    let status = child.wait().expect("reap killed child");
    reader.join().expect("join child output reader");
    assert!(
        !status.success(),
        "child should not exit normally after the forced kill"
    );

    let sqlite_path = directory.path().join("mnexis.db");
    let connection = open_sqlite(&sqlite_path).expect("reopen SQLite after process kill");
    let sqlite_value: String = connection
        .query_row(
            "SELECT value FROM crash_markers WHERE id = ?1",
            [SQLITE_MARKER],
            |row| row.get(0),
        )
        .expect("read committed SQLite marker after restart");
    assert_eq!(sqlite_value, "committed-before-kill");

    let vector_uri = directory
        .path()
        .join("lancedb")
        .to_string_lossy()
        .into_owned();
    let db = lancedb::connect(vector_uri.as_str())
        .execute()
        .await
        .expect("reopen LanceDB after process kill");
    let table = db
        .open_table(TABLE)
        .execute()
        .await
        .expect("open persisted LanceDB table");
    let rows = table
        .count_rows(Some(format!("id = '{LANCE_MARKER}'")))
        .await
        .expect("count persisted LanceDB marker");
    assert_eq!(
        rows, 1,
        "the acknowledged LanceDB row should survive restart"
    );
}

fn marker_batch(id: &str) -> RecordBatch {
    let schema = Arc::new(Schema::new(vec![
        Field::new("id", DataType::Utf8, false),
        Field::new(
            "vector",
            DataType::FixedSizeList(Arc::new(Field::new("item", DataType::Float32, true)), 3),
            false,
        ),
    ]));
    let ids: ArrayRef = Arc::new(StringArray::from(vec![id]));
    let vectors: ArrayRef = Arc::new(
        FixedSizeListArray::from_iter_primitive::<Float32Type, _, _>(
            [Some(vec![Some(0.1), Some(0.2), Some(0.3)])],
            3,
        ),
    );
    RecordBatch::try_new(schema, vec![ids, vectors]).expect("build LanceDB marker batch")
}
